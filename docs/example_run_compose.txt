# Mismo codigo que la UI (AiServiceClient.rag_estimate + db.save_estimation), lanzado dentro del contenedor business-backend

$ docker compose exec -T business-backend python - < e2e.py
    | # Runs inside business-backend: same code path as the UI's RAG mode.
    | import time
    | 
    | import db
    | from ai_client import AiServiceClient
    | 
    | transcript = open("examples/transcripts/01_clear.txt", encoding="utf-8").read()
    | t0 = time.perf_counter()
    | est = AiServiceClient().rag_estimate(transcript)
    | elapsed = time.perf_counter() - t0
    | 
    | print(f"confidence={est['confidence']} total_engineer_days={est.get('total_engineer_days')} "
    |       f"duration_weeks={est.get('duration_weeks')} ({elapsed:.1f}s)")
    | for c in est.get("cost_breakdown", [])[:8]:
    |     print(f"  - {c['name']}: {c['engineer_days']} dias, fuentes={c['sources']}")
    | print("fuentes citadas:", [s["source_id"] for s in est.get("sources", [])])
    | 
    | conn = db.connect()
    | db.save_estimation(conn, mode="rag", input_text=transcript, result_json=est, elapsed_s=elapsed)
    | n = conn.execute("select count(*) as n from estimations").fetchone()["n"]
    | print(f"guardada en postgres; historial: {n} estimaciones")
